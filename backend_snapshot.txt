# ============================================================
# FIND KEYWORD MATCHES
# ============================================================

def find_matches(text, keyword_dictionary):

    matches = []

    text = str(text).lower()

    for category, keywords in keyword_dictionary.items():

        for keyword in keywords:

            if keyword in text:
                matches.append(category)
                break

    return matches


# ============================================================
# EXTRACT ENTITIES FROM ONE REPORT
# ============================================================

def extract_entities(text):

    if not isinstance(text, str):
        text = ""

    hazards = find_matches(text, HAZARD_KEYWORDS)

    energies = find_matches(text, ENERGY_KEYWORDS)

    barriers = find_matches(text, BARRIER_KEYWORDS)

    return {
        "hazards": hazards,
        "energy": energies,
        "barriers": barriers
    }


# ============================================================
# APPLY EXTRACTION TO DATASET
# ============================================================

def extract_from_dataframe(df):

    df = df.copy()

    results = df["clean_text"].apply(extract_entities)

    df["extracted_hazard"] = results.apply(
        lambda x: ", ".join(x["hazards"])
    )

    df["extracted_energy"] = results.apply(
        lambda x: ", ".join(x["energy"])
    )

    df["extracted_barrier"] = results.apply(
        lambda x: ", ".join(x["barriers"])
    )

    return df

import pandas as pd

# ============================================================
# SIF PRECURSOR PATTERN ANALYSIS
# ============================================================

def analyze_patterns(df):

    df = df.copy()

    # --------------------------------------------------------
    # Convert date
    # --------------------------------------------------------

    df["date"] = pd.to_datetime(df["date"], errors="coerce")

    # --------------------------------------------------------
    # 1. SIF patterns by site
    # --------------------------------------------------------

    site_patterns = (
        df.groupby("site")
        .agg(
            total_reports=("report_id", "count"),
            sif_reports=("predicted_sif", lambda x: (x == "YES").sum()),
            average_risk=("risk_score", "mean")
        )
        .reset_index()
    )

    site_patterns["sif_rate"] = (
        site_patterns["sif_reports"]
        / site_patterns["total_reports"]
        * 100
    ).round(2)

    # --------------------------------------------------------
    # 2. SIF patterns by location
    # --------------------------------------------------------

    location_patterns = (
        df.groupby("location")
        .agg(
            total_reports=("report_id", "count"),
            sif_reports=("predicted_sif", lambda x: (x == "YES").sum()),
            average_risk=("risk_score", "mean")
        )
        .reset_index()
    )

    location_patterns["sif_rate"] = (
        location_patterns["sif_reports"]
        / location_patterns["total_reports"]
        * 100
    ).round(2)

    # --------------------------------------------------------
    # 3. SIF patterns by activity
    # --------------------------------------------------------

    activity_patterns = (
        df.groupby("activity")
        .agg(
            total_reports=("report_id", "count"),
            sif_reports=("predicted_sif", lambda x: (x == "YES").sum()),
            average_risk=("risk_score", "mean")
        )
        .reset_index()
    )

    activity_patterns["sif_rate"] = (
        activity_patterns["sif_reports"]
        / activity_patterns["total_reports"]
        * 100
    ).round(2)

    # --------------------------------------------------------
    # 4. Risk-level distribution
    # --------------------------------------------------------

    risk_distribution = (
        df["risk_level"]
        .value_counts()
        .reset_index()
    )

    risk_distribution.columns = ["risk_level", "report_count"]

    # --------------------------------------------------------
    # 5. Precursor indicator frequency
    # --------------------------------------------------------

    indicator_counts = {}

    for indicators in df["sif_indicators"].fillna(""):

        if not indicators:
            continue

        for indicator in indicators.split(", "):

            if indicator:
                indicator_counts[indicator] = (
                    indicator_counts.get(indicator, 0) + 1
                )

    indicator_patterns = pd.DataFrame(
        list(indicator_counts.items()),
        columns=["indicator", "count"]
    )

    if not indicator_patterns.empty:
        indicator_patterns = indicator_patterns.sort_values(
            "count",
            ascending=False
        )

    # --------------------------------------------------------
    # 6. Primary risk frequency
    # --------------------------------------------------------

    primary_risk_patterns = (
        df["primary_risk"]
        .value_counts()
        .reset_index()
    )

    primary_risk_patterns.columns = [
        "primary_risk",
        "count"
    ]

    # --------------------------------------------------------
    # 7. Barrier failure frequency
    # --------------------------------------------------------

    barrier_counts = {}

    for barriers in df["barrier_failures"].fillna(""):

        if not barriers:
            continue

        for barrier in barriers.split(", "):

            if barrier:
                barrier_counts[barrier] = (
                    barrier_counts.get(barrier, 0) + 1
                )

    barrier_patterns = pd.DataFrame(
        list(barrier_counts.items()),
        columns=["barrier_failure", "count"]
    )

    if not barrier_patterns.empty:
        barrier_patterns = barrier_patterns.sort_values(
            "count",
            ascending=False
        )

    return {
        "site_patterns": site_patterns,
        "location_patterns": location_patterns,
        "activity_patterns": activity_patterns,
        "risk_distribution": risk_distribution,
        "indicator_patterns": indicator_patterns,
        "primary_risk_patterns": primary_risk_patterns,
        "barrier_patterns": barrier_patterns
    }

import re
import pandas as pd


ABBREVIATIONS = {
    "loto": "lockout tagout",
    "ptw": "permit to work",
    "ppe": "personal protective equipment",
    "lof": "line of fire",
}


def clean_text(text):
    """
    Clean and normalize safety report text.
    """

    if pd.isna(text):
        return ""

    text = str(text).lower()

    for abbreviation, replacement in ABBREVIATIONS.items():
        text = text.replace(abbreviation, replacement)
    text = text.replace("nearthe", "near the")
    text = text.replace("noticednear", "noticed near")
    text = text.replace("supplyremained", "supply remained")
    text = text.replace("equipmentwhile", "equipment while")

    text = re.sub(r"[^a-z0-9\s]", " ", text)
    text = re.sub(r"\s+", " ", text)
    return text.strip()


def preprocess_dataframe(df):
    """
    Add cleaned text to the dataframe.
    """

    if "free_text" not in df.columns:
        raise ValueError(
            "The dataframe must contain a 'free_text' column."
        )

    df = df.copy()

    df["clean_text"] = df["free_text"].apply(clean_text)

    return df


def load_and_preprocess(csv_path):
    """
    Load CSV and preprocess the safety reports.
    """

    df = pd.read_csv(csv_path)

    return preprocess_dataframe(df)

import pandas as pd


# ============================================================
# RISK SCORING CONFIGURATION
# ============================================================

HIGH_RISK_INDICATORS = {
    "Electrical Energy",
    "Confined Space",
    "Working at Height",
    "Pressure",
    "Mechanical Energy",
    "Line of Fire",
    "Vehicle",
    "Hot Work",
    "Safety Control Bypass",
    "Work Authorization"
}


# ============================================================
# BARRIER FAILURE INDICATORS
# ============================================================

BARRIER_FAILURE_KEYWORDS = [
    "without",
    "missing",
    "not completed",
    "not confirmed",
    "not used",
    "not provided",
    "not in place",
    "remained connected",
    "overridden",
    "overrode",
    "defeated",
    "bypassed",
    "without proper",
    "without required",
    "required authorization was missing"
]


# ============================================================
# DETECT BARRIER FAILURE
# ============================================================

def detect_barrier_failure(text):

    text = str(text).lower()

    matched = []

    for keyword in BARRIER_FAILURE_KEYWORDS:

        if keyword in text:
            matched.append(keyword)

    return matched


# ============================================================
# CALCULATE RISK
# ============================================================

def calculate_risk(
    text,
    predicted_sif,
    sif_score,
    sif_indicators,
    extracted_hazard="",
    extracted_energy="",
    extracted_barrier=""
):

    text = str(text).lower()

    # --------------------------------------------------------
    # CONVERT SIF INDICATORS INTO LIST
    # --------------------------------------------------------

    if isinstance(sif_indicators, str):

        indicators = [
            x.strip()
            for x in sif_indicators.split(",")
            if x.strip()
        ]

    elif isinstance(sif_indicators, list):

        indicators = sif_indicators

    else:

        indicators = []

    # --------------------------------------------------------
    # INITIAL SCORE
    # --------------------------------------------------------

    score = 0
    risk_factors = []

    # --------------------------------------------------------
    # SIF PRECURSOR DETECTED
    # --------------------------------------------------------

    if predicted_sif == "YES":

        score += 40

        risk_factors.append(
            "SIF precursor detected"
        )

    # --------------------------------------------------------
    # HIGH-RISK INDICATOR
    # --------------------------------------------------------

    high_risk_matches = [
        indicator
        for indicator in indicators
        if indicator in HIGH_RISK_INDICATORS
    ]

    if high_risk_matches:

        score += 20

        risk_factors.append(
            "High-risk indicator: "
            + ", ".join(high_risk_matches)
        )

    # --------------------------------------------------------
    # HAZARDOUS ENERGY
    # --------------------------------------------------------

    if extracted_energy:

        score += 15

        risk_factors.append(
            "Hazardous energy identified"
        )

    # --------------------------------------------------------
    # BARRIER FAILURE
    # --------------------------------------------------------

    barrier_failures = detect_barrier_failure(text)

    if barrier_failures:

        score += 15

        risk_factors.append(
            "Barrier/control failure detected"
        )

    # --------------------------------------------------------
    # MULTIPLE SIF INDICATORS
    # --------------------------------------------------------

    if len(indicators) >= 2:

        score += 10

        risk_factors.append(
            "Multiple SIF precursor indicators"
        )

    # --------------------------------------------------------
    # MAXIMUM SCORE
    # --------------------------------------------------------

    score = min(score, 100)

    # --------------------------------------------------------
    # RISK LEVEL
    # --------------------------------------------------------

    if score >= 80:

        risk_level = "CRITICAL"

    elif score >= 60:

        risk_level = "HIGH"

    elif score >= 30:

        risk_level = "MEDIUM"

    else:

        risk_level = "LOW"

    # --------------------------------------------------------
    # PRIMARY RISK
    # --------------------------------------------------------

    if high_risk_matches:

        primary_risk = high_risk_matches[0]

    elif extracted_hazard:

        primary_risk = extracted_hazard

    else:

        primary_risk = "General Safety Risk"

    # --------------------------------------------------------
    # EXPLANATION
    # --------------------------------------------------------

    if risk_factors:

        reason = "; ".join(risk_factors)

    else:

        reason = "No significant precursor risk detected"

    # --------------------------------------------------------
    # RETURN RESULT
    # --------------------------------------------------------

    return {
        "risk_score": score,
        "risk_level": risk_level,
        "primary_risk": primary_risk,
        "risk_factors": risk_factors,
        "barrier_failures": barrier_failures,
        "risk_reason": reason
    }


# ============================================================
# APPLY RISK SCORING TO DATAFRAME
# ============================================================

def apply_risk_scoring(df):

    df = df.copy()

    results = df.apply(
        lambda row: calculate_risk(
            row["clean_text"],
            row["predicted_sif"],
            row["sif_score"],
            row["sif_indicators"],
            row["extracted_hazard"],
            row["extracted_energy"],
            row["extracted_barrier"]
        ),
        axis=1
    )

    # --------------------------------------------------------
    # STORE RESULTS
    # --------------------------------------------------------

    df["risk_score"] = results.apply(
        lambda x: x["risk_score"]
    )

    df["risk_level"] = results.apply(
        lambda x: x["risk_level"]
    )

    df["primary_risk"] = results.apply(
        lambda x: x["primary_risk"]
    )

    df["risk_factors"] = results.apply(
        lambda x: ", ".join(x["risk_factors"])

# ============================================================
# SIF DETECTION
# ============================================================

def detect_sif(text, extracted_hazard="", extracted_energy=""):

    text = str(text).lower()

    matched_indicators = []

    for category, keywords in SIF_INDICATORS.items():

        for keyword in keywords:

            if keyword in text:
                matched_indicators.append(category)
                break

    # --------------------------------------------------------
    # RISK SCORING
    # --------------------------------------------------------

    score = 0

    # High-energy extracted entity
    if extracted_energy:
        score += 40

    # High-risk SIF precursor
    high_risk_matches = [
        indicator
        for indicator in matched_indicators
        if indicator in HIGH_RISK_INDICATORS
    ]

    if high_risk_matches:
        score += 60

    # Fire indicator
    if "Fire" in matched_indicators:
        score += 40

    # Multiple independent indicators
    if len(matched_indicators) >= 2:
        score += 20

    # Maximum score = 100
    score = min(score, 100)

    # --------------------------------------------------------
    # CLASSIFICATION
    # --------------------------------------------------------

    if score >= 60:
        prediction = "YES"
    else:
        prediction = "NO"

    return {
        "sif_prediction": prediction,
        "sif_score": score,
        "sif_indicators": matched_indicators
    }


# ============================================================
# APPLY TO DATASET
# ============================================================

def apply_sif_detection(df):

    df = df.copy()

    results = df.apply(
        lambda row: detect_sif(
            row["clean_text"],
            row["extracted_hazard"],
            row["extracted_energy"]
        ),
        axis=1
    )

    df["predicted_sif"] = results.apply(
        lambda x: x["sif_prediction"]
    )

    df["sif_score"] = results.apply(
        lambda x: x["sif_score"]
    )

    df["sif_indicators"] = results.apply(
        lambda x: ", ".join(x["sif_indicators"])
    )

    return df
